async function salvarColaborador() { const idEdit = document.getElementById('edit-id').value; const nomeColab = document.getElementById('input-nome-colab').value.trim(); const funcaoSelecionada = document.getElementById('novo-funcao').value; if (!nomeColab) { alert("⚠️ Por favor, insira o nome do colaborador."); return; } if (idEdit) { await supabaseClient.from('colaboradores').update({ nome: nomeColab }).eq('id', idEdit); const colab = colaboradores.find(c => c.id === idEdit); if (colab && !isColabEmDSRHoje(colab)) { await atualizarDiaSupabase(idEdit, diaHoje, funcaoSelecionada); } else if (colab && isColabEmDSRHoje(colab) && isStatusEspecial(funcaoSelecionada)) { await atualizarDiaSupabase(idEdit, diaHoje, funcaoSelecionada); } } else { const escalaColab = document.getElementById('select-escala').value; const dsrColab = document.getElementById('select-dsr').value; const horarioColab = document.getElementById('input-horario').value; const { data, error } = await supabaseClient.from('colaboradores').insert([{ nome: nomeColab, escala: escalaColab, dsr: dsrColab, horario: horarioColab }]).select(); if (!error && data && data.length > 0) { const dataHoje = new Date(anoAtual, mesAtualIndex, diaHoje); const hojeEhDSR = isDiaDSR(dsrColab, dataHoje); if (!hojeEhDSR || isStatusEspecial(funcaoSelecionada)) { await atualizarDiaSupabase(data[0].id, diaHoje, funcaoSelecionada); } } else if (error) { alert("Erro ao salvar: " + error.message); } } fecharModalAdd(); await carregarDadosSupabase(); renderizar(); }
