function renderizar() { const termoBuscaNorm = normalizarTexto(document.getElementById('input-busca').value); const containerSetores = document.getElementById('container-setores'); containerSetores.innerHTML = ""; // Agrupa const grupos = { operacao: [], dsr: [], falta: [], atestado: [] }; colaboradores.forEach(c => { const tipo = obterStatusHoje(c); if (tipo === 'operacao') grupos.operacao.push(c); else if (tipo === 'dsr') grupos.dsr.push(c); else if (tipo === 'falta') grupos.falta.push(c); else if (tipo === 'atestado') grupos.atestado.push(c); }); const filtrar = lista => lista.filter(c => matchBuscaAvancada(termoBuscaNorm, c.nome, c.escala, c.dsr, c.horario) ); const operacaoF = filtrar(grupos.operacao); const dsrF = filtrar(grupos.dsr); const faltaF = filtrar(grupos.falta); const atestadoF = filtrar(grupos.atestado); // KPIs e badges document.getElementById('kpi-total').innerText = colaboradores.length; document.getElementById('kpi-operacao').innerText = grupos.operacao.length; document.getElementById('kpi-dsr').innerText = grupos.dsr.length; document.getElementById('kpi-faltas').innerText = grupos.falta.length; document.getElementById('kpi-atestados').innerText = grupos.atestado.length; const badges = { 'badge-todos': colaboradores.length, 'badge-operacao': grupos.operacao.length, 'badge-dsr': grupos.dsr.length, 'badge-faltas': grupos.falta.length, 'badge-atestados': grupos.atestado.length }; Object.entries(badges).forEach(([id, val]) => { const el = document.getElementById(id); if (el) el.innerText = val; const elM = document.getElementById(id + '-m'); if (elM) elM.innerText = val; }); // ===================================================================== // MODO BUSCA ATIVA — sobe os resultados pro topo // ===================================================================== if (termoBuscaNorm) { const todosMatch = [ ...dsrF.map(c => ({ c, tipo: 'dsr' })), ...faltaF.map(c => ({ c, tipo: 'falta' })), ...atestadoF.map(c => ({ c, tipo: 'atestado' })), ...operacaoF.map(c => ({ c, tipo: 'operacao' })) ]; // Aplica o filtro ativo (se houver) const filtroMap = { 'todos': null, 'operacao': 'operacao', 'dsr': 'dsr', 'faltas': 'falta', 'atestados': 'atestado' }; const tipoPermitido = filtroMap[filtroAtivo]; const filtrados = tipoPermitido ? todosMatch.filter(x => x.tipo === tipoPermitido) : todosMatch; if (filtrados.length === 0) { containerSetores.innerHTML = `

Nenhum resultado para "${document.getElementById('input-busca').value}".

`; lucide.createIcons(); return; } // Bloco destacado de resultados const blocoResultados = document.createElement('div'); blocoResultados.className = "bg-amber-950/10 border-2 border-amber-500/40 rounded-2xl p-4 shadow-2xl"; blocoResultados.innerHTML = `

Resultados da busca — "${document.getElementById('input-busca').value}"

${filtrados.length} encontrado${filtrados.length === 1 ? '' : 's'}
${filtrados.map(({ c, tipo }) => cardColaborador(c, tipo)).join('')}
`; containerSetores.appendChild(blocoResultados); lucide.createIcons(); return; } // ===================================================================== // MODO NORMAL — sem busca // ===================================================================== const blocoDSR = criarBloco('Colaboradores em DSR Hoje', '— não podem ser escalados', dsrF, 'orange', '🌴', `${dsrF.length} em descanso`); const blocoFalta = criarBloco('Faltas Hoje', '— ausências não justificadas', faltaF, 'yellow', '⚠️', `${faltaF.length} falta${faltaF.length === 1 ? '' : 's'}`); const blocoAtestado = criarBloco('Atestados Hoje', '— ausências justificadas', atestadoF, 'red', '🏥', `${atestadoF.length} atestado${atestadoF.length === 1 ? '' : 's'}`); if (filtroAtivo === 'todos') { if (blocoDSR) containerSetores.appendChild(blocoDSR); if (blocoFalta) containerSetores.appendChild(blocoFalta); if (blocoAtestado) containerSetores.appendChild(blocoAtestado); renderizarSetores(operacaoF, containerSetores); } else if (filtroAtivo === 'operacao') { renderizarSetores(operacaoF, containerSetores); } else if (filtroAtivo === 'dsr') { if (blocoDSR) containerSetores.appendChild(blocoDSR); else containerSetores.innerHTML = `

Nenhum colaborador em DSR hoje.

`; } else if (filtroAtivo === 'faltas') { if (blocoFalta) containerSetores.appendChild(blocoFalta); else containerSetores.innerHTML = `

Nenhuma falta registrada hoje.

`; } else if (filtroAtivo === 'atestados') { if (blocoAtestado) containerSetores.appendChild(blocoAtestado); else containerSetores.innerHTML = `

Nenhum atestado registrado hoje.

`; } lucide.createIcons(); }
