        function realizarRodizioGeral() {

            colaboradores.forEach(c => {

                const ultimo = c.historicoDias.length > 0 ? c.historicoDias[c.historicoDias.length - 1] : FUNCOES_LOGISTICA[0];

                const indexAtual = FUNCOES_LOGISTICA.indexOf(ultimo);

                const proximoIndex = indexAtual !== -1 ? (indexAtual + 1) % FUNCOES_LOGISTICA.length : 0;

                c.historicoDias.push(FUNCOES_LOGISTICA[proximoIndex]);

            });

            renderizar();

            alert('Rodízio executado! Todos os colaboradores avançaram para o próximo dia e função.');

        }
